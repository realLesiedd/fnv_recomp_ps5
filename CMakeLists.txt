cmake_minimum_required(VERSION 3.20)
project(fnv_recomp C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# Generated from the player's own FalloutNV.exe by tools/recomp.py (never committed).
file(GLOB GEN_SOURCES ${CMAKE_SOURCE_DIR}/gen/funcs_*.c)
add_library(fnv_gen OBJECT ${GEN_SOURCES} gen/table.c gen/image.c)
# -O1 keeps compile time reasonable for ~370 MB of generated C; locals still get registers.
set_source_files_properties(${GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "-O1;-g0;-w;-fno-strict-aliasing;-fwrapv")
target_include_directories(fnv_gen PRIVATE rt)

file(GLOB RT_SOURCES ${CMAKE_SOURCE_DIR}/rt/*.c)
list(REMOVE_ITEM RT_SOURCES ${CMAKE_SOURCE_DIR}/rt/platform_sdl.c ${CMAKE_SOURCE_DIR}/rt/platform_null.c)
if(PS5)
  # jailbroken PS5 payload (ps5-payload-sdk): cmake -DCMAKE_TOOLCHAIN_FILE=$PS5_PAYLOAD_SDK/toolchain/prospero.cmake
  add_compile_options(-fno-omit-frame-pointer)
endif()
find_package(SDL2 QUIET)
if(SDL2_FOUND AND NOT FNV_HEADLESS)
  list(APPEND RT_SOURCES rt/platform_sdl.c)
else()
  list(APPEND RT_SOURCES rt/platform_null.c)
endif()
# GPU renderer: DXVK (Direct3D 9 on Vulkan). Point FNV_DXVK_DIR at a DXVK native install prefix.
set(FNV_DXVK_DIR "" CACHE PATH "DXVK native install prefix (enables the GPU renderer)")
list(REMOVE_ITEM RT_SOURCES ${CMAKE_SOURCE_DIR}/rt/hostd3d9_null.c)
if(FNV_DXVK_DIR AND EXISTS "${FNV_DXVK_DIR}/include/dxvk/d3d9.h")
  list(APPEND RT_SOURCES rt/hostd3d9.cpp)
  set(FNV_HAVE_DXVK ON)
else()
  list(APPEND RT_SOURCES rt/hostd3d9_null.c)
endif()
add_executable(fnv ${RT_SOURCES} $<TARGET_OBJECTS:fnv_gen>)
if(FNV_HAVE_DXVK)
  target_include_directories(fnv PRIVATE ${FNV_DXVK_DIR}/include/dxvk)
  file(GLOB DXVK_LIB "${FNV_DXVK_DIR}/lib*/*/libdxvk_d3d9.so" "${FNV_DXVK_DIR}/lib*/libdxvk_d3d9.so" "${FNV_DXVK_DIR}/lib*/*/libdxvk_d3d9.a" "${FNV_DXVK_DIR}/lib*/libdxvk_d3d9.a")
  list(GET DXVK_LIB 0 DXVK_LIB)
  target_link_libraries(fnv PRIVATE ${DXVK_LIB})
  get_filename_component(DXVK_LIB_DIR ${DXVK_LIB} DIRECTORY)
  set_target_properties(fnv PROPERTIES BUILD_RPATH ${DXVK_LIB_DIR})
  message(STATUS "GPU renderer: DXVK from ${DXVK_LIB}")
endif()
if(SDL2_FOUND AND NOT FNV_HEADLESS)
  # Linux: the shared SDL, which DXVK also loads (two SDL copies would not share the window or video state)
  if(PS5 AND TARGET SDL2::SDL2-static)
    target_link_libraries(fnv PRIVATE SDL2::SDL2-static)
  else()
    target_link_libraries(fnv PRIVATE SDL2::SDL2)
  endif()
endif()
if(PS5)
  set_target_properties(fnv PROPERTIES OUTPUT_NAME fnv SUFFIX .elf)
  target_link_libraries(fnv PRIVATE SceVideoOut ScePad SceAudioOut SceUserService SceSystemService kernel_sys)
endif()
target_include_directories(fnv PRIVATE rt)
target_compile_options(fnv PRIVATE -Wall -Wno-unused-function -fno-strict-aliasing -fwrapv $<$<COMPILE_LANGUAGE:CXX>:-Wno-unused-variable>)
# the app's own assets (button prompts), built into the executable
file(GLOB_RECURSE ASSET_FILES ${CMAKE_SOURCE_DIR}/assets/*)
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/assets_data.c
  COMMAND python3 ${CMAKE_SOURCE_DIR}/tools/embed_assets.py ${CMAKE_SOURCE_DIR}/assets ${CMAKE_BINARY_DIR}/assets_data.c
  DEPENDS ${ASSET_FILES} ${CMAKE_SOURCE_DIR}/tools/embed_assets.py)
target_sources(fnv PRIVATE ${CMAKE_BINARY_DIR}/assets_data.c)

# Bink movies: FFmpeg's Bink decoders (libavcodec, LGPL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil)
target_link_libraries(fnv PRIVATE PkgConfig::FFMPEG)
target_link_libraries(fnv PRIVATE m pthread)
if(NOT PS5)
  # export only the recompiled functions (crash backtraces name them); exporting the
  # runtime's own symbols lets libraries loaded later (Vulkan drivers, SDL) bind to them
  target_link_options(fnv PRIVATE -Wl,--dynamic-list=${CMAKE_SOURCE_DIR}/rt/exports.list)
endif()
